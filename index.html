Google Sheets (через Apps Script)
Інструкція як налаштувати Apps Script — нижче.
Інструкція: Google Sheets Apps Script

1. Відкрийте Google Sheets → Розширення → Apps Script
2. Вставте код нижче та збережіть
3. Розгорніть як Веб-додаток (Deploy → New deployment → Web app → хто має доступ: всі)
4. Скопіюйте URL і вставте вище

function doPost(e) { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // Аркуш "Cars" if (data.sheet === 'cars') { const sh = ss.getSheetByName('Cars') || ss.insertSheet('Cars'); if (sh.getLastRow() === 0) sh.appendRow(['Власник','Марка','Модель','Рік','Пробіг','Двигун','Пальне','VIN','Номер','Нотатки','Дата']); sh.appendRow([data.owner,data.make,data.model,data.year, data.mileage,data.engine,data.fuel,data.vin,data.plate||'',data.notes||'',new Date()]); } // Аркуш "Maintenance" if (data.sheet === 'maintenance') { const sh = ss.getSheetByName('Maintenance') || ss.insertSheet('Maintenance'); if (sh.getLastRow() === 0) sh.appendRow(['Власник','Тип роботи','Пробіг виконання','Дата запису']); data.items.forEach(item => { sh.appendRow([data.owner, item.type, item.mileage, new Date()]); }); } // Аркуш "ServiceLogs" if (data.sheet === 'service_log') { const sh = ss.getSheetByName('ServiceLogs') || ss.insertSheet('ServiceLogs'); if (sh.getLastRow() === 0) sh.appendRow(['Власник','Дата','Пробіг','СТО','Роботи','Запчастини','Сума']); sh.appendRow([data.owner,data.date,data.mileage,data.station, data.works,data.parts,data.cost]); } return ContentService.createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); }
